package main

import (
	"encoding/json"
	"fmt"
	"os"
	"time"
)

type verifyCase struct {
	Secret    string `json:"secret"`
	Body      string `json:"body"`
	Timestamp string `json:"timestamp"`
	Signature string `json:"signature"`
	Now       int64  `json:"now"`
}

func main() {
	var input verifyCase
	if err := json.NewDecoder(os.Stdin).Decode(&input); err != nil {
		fmt.Println("ERROR:", err)
		return
	}
	if err := VerifyEventsSignature(input.Secret, []byte(input.Body), input.Timestamp, input.Signature, time.Unix(input.Now, 0)); err != nil {
		fmt.Println("ERROR:", err)
		return
	}
	fmt.Println("OK")
}
